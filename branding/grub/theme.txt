# Osiris boot menu.
#
# Global properties use "key: value"; components use "+ name { key = value }".
# Fonts are referenced by the family name baked into the .pf2 file, which
# provision/grub.py sets with grub-mkfont -n. A name GRUB cannot find falls
# back to a bitmap font silently, so the two have to agree.

title-text: ""
desktop-image: "background.png"
desktop-color: "#0d0d10"
terminal-font: "Osiris Mono 16"
terminal-left: "8%"
terminal-top: "12%"
terminal-width: "84%"
terminal-height: "76%"

# The mark, above the menu. Its own component rather than part of the
# background, so it stays crisp if GRUB has to scale the backdrop.

    top = 14%
    left = 50%-45
    width = 91
    height = 220
}

    top = 14%+236
